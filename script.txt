local HttpService = game:GetService("HttpService")

if not getgc then print("нет getgc"); return end
print("[PIN] Сканируем GC...")

local curHash = nil
local gc = getgc(true)
local n = 0
for _, v in pairs(gc) do
    n = n+1
    if n % 1500 == 0 then task.wait(0.05) end
    if type(v) == "table" then
        pcall(function()
            local val = rawget(v, "_value")
            if type(val)=="string" and #val==64 and val:match("^[0-9a-fA-F]+$") then
                if rawget(v,"__conn")~=nil or rawget(v,"_dependents")~=nil then
                    curHash = val:lower()
                end
            end
            local ps = rawget(v, "TradePINCode")
            if type(ps)=="table" then
                local h = rawget(ps, "_value")
                if type(h)=="string" and #h==64 and h:match("^[0-9a-fA-F]+$") then
                    curHash = h:lower()
                end
            end
        end)
        if curHash then break end
    end
end

if not curHash then print("[PIN] Хеш не найден"); return end
print("[PIN] Хеш получен, качаем pins.txt...")

local ok, data = pcall(function()
    return game:HttpGet("https://raw.githubusercontent.com/mrkenderlike/BBpins/refs/heads/main/pins.txt")
end)
if not ok or not data or data == "" then print("[PIN] Не удалось скачать pins.txt"); return end
print("[PIN] Список загружен, ищем совпадение...")

local found = nil
for line in data:gmatch("[^\r\n]+") do
    local pin, hash = line:match("^(%d+),([0-9a-fA-F]+)")
    if pin and hash and hash:lower() == curHash then
        found = pin
        break
    end
end

if found then
    print("[PIN] ВАШ ПИН: " .. found)
else
    print("[PIN] Пин не найден в таблице")
end
